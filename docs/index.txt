import { app, shell, BrowserWindow, ipcMain } from 'electron'
import { path, join } from 'path'
import { electronApp, optimizer, is } from '@electron-toolkit/utils'
import { eq } from 'drizzle-orm' 

import { getDb } from './db/client.js'
import {
  chambre,
  client,
  consommation,
  demande,
  hotel,
  reservation,
  service,

} from './db/schema/schema.js'

let onboardingWindow = null
let loginWindow = null
let mainWindow = null

const db = getDb()

function loadRenderer(win, flag = null) {
  if (is.dev && process.env['ELECTRON_RENDERER_URL']) {
    const base = process.env['ELECTRON_RENDERER_URL']
    win.loadURL(flag ? `${base}?${flag}=1` : base)
  } else {
    win.loadFile(join(__dirname, '../renderer/index.html'), flag ? { query: { [flag]: '1' } } : {})
  }
}

function createOnboardingWindow() {
  onboardingWindow = new BrowserWindow({
    width: 720, height: 300, frame: false, resizable: false, maximizable: false, fullscreenable: false, icon: path.join(__dirname, '../../assets/icon.png'),
    webPreferences: { preload: join(__dirname, '../preload/index.js'), sandbox: false }
  })
  loadRenderer(onboardingWindow, 'onboarding')
  onboardingWindow.on('closed', () => { onboardingWindow = null })
}

function createLoginWindow() {
  loginWindow = new BrowserWindow({
    width: 1000, height: 650, resizable: false, maximizable: false, fullscreenable: false, frame: false, icon: path.join(__dirname, '../../assets/icon.png'),
    webPreferences: { preload: join(__dirname, '../preload/index.js'), sandbox: false }
  })
  loadRenderer(loginWindow, 'login')
  loginWindow.on('closed', () => { loginWindow = null })
}

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1080, height: 670, fullscreenable: true, maximizable: true, frame: false, icon: path.join(__dirname, '../../assets/icon.png'),
    backgroundColor: '#d9cde3', titleBarStyle: 'hidden', resizable: true, minimizable: true,
    titleBarOverlay: { color: '#000000', symbolColor: '#1A1A1A', height: 32 },
    webPreferences: { preload: join(__dirname, '../preload/index.js'), sandbox: false }
  })
  mainWindow.on('ready-to-show', () => { mainWindow.maximize(); mainWindow.show() })
  mainWindow.on('closed', () => { mainWindow = null })
  mainWindow.webContents.setWindowOpenHandler((details) => { shell.openExternal(details.url); return { action: 'deny' } })
  mainWindow.maximize()
  loadRenderer(mainWindow)
}

ipcMain.on('window:minimize', (e) => BrowserWindow.fromWebContents(e.sender)?.minimize())
ipcMain.on('window:close', (e) => BrowserWindow.fromWebContents(e.sender)?.close())
ipcMain.on('window:maximize', (e) => BrowserWindow.fromWebContents(e.sender)?.maximize())
ipcMain.on('onboarding:done', () => { if (!loginWindow && !mainWindow) createLoginWindow(); if (onboardingWindow) onboardingWindow.close() })
ipcMain.on('login:done', () => { if (!mainWindow) createWindow(); if (loginWindow) loginWindow.close() })

app.whenReady().then(() => {
  electronApp.setAppUserModelId('com.hypnos.app')
  app.on('browser-window-created', (_, window) => { optimizer.watchWindowShortcuts(window) })
  createOnboardingWindow()
  app.on('activate', () => { if (BrowserWindow.getAllWindows().length === 0) createOnboardingWindow() })
})
app.on('window-all-closed', () => { if (process.platform !== 'darwin') app.quit() })

// CREATE - Tu avais bon, je corrige juste chambre
ipcMain.handle('clients:create', async (_, data) => {
  const newClient = { nom: data.nom, prenom: data.prenom, tel: data.tel, email: data.email, sexe: data.sexe, document: data.document, preference: data.preference, loyalty: data.loyalty }
  const result = await db.insert(client).values(newClient).returning()
  return result[0]
})
ipcMain.handle('typechambre:create', async (_, data) => {
  const newTypeRoom = { remoteId: data.remoteId, label: data.label, prix: data.prix, capacite: data.capacite }
  const result = await db.insert(typeDeChambre).values(newTypeRoom).returning()
  return result[0]
})
ipcMain.handle('chambre:create', async (_, data) => {
  if (!data.typeId) throw new Error('Il faut choisir un type de chambre.')
  const typeExiste = await db.query.typeDeChambre.findFirst({ where: (t, { eq }) => eq(t.id, Number(data.typeId)) })
  if (!typeExiste) throw new Error("Ce type de chambre n'existe pas")
  const newRoom = { remoteId: data.remoteId, label: data.label, numero: data.numero, typeId: Number(data.typeId), etage: data.etage, statut: data.statut }
  const result = await db.insert(chambre).values(newRoom).returning()
  return result[0]
})

// UPDATE - TOUTES TES TABLES EN UNE FOIS
function miseAJour(pseudoDuChanel, message, table) {
  ipcMain.handle(`${pseudoDuChanel}:update`, async (event, id, data) => {
    console.log(message, id)
    const { id: _id, ...dataToUpdate } = data // on n'écrase pas l'id
    const result = await db.update(table).set(dataToUpdate).where(eq(table.id, Number(id))).returning()
    return result[0]
  })
}

function suppression(pseudoDuChanel, message, table) {
