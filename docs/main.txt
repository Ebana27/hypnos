// App.jsx

import { useState } from 'react'
import { HashRouter, Route, Routes, Navigate, useLocation } from 'react-router-dom'

// Composants Pages
import LoginRoute from './components/Login'
import Register from './components/Register'
import Sidebar from './components/base/SideBar'
import TiltleBar from './components/base/TiltleBar'

// Pages react
import AccountingPage from './pages/AccountingPage'
import BookingPage from './pages/BookingPage'
import ClientPage from './pages/ClientPage'
import CommunicationPage from './pages/CommunicationPage'
import HomePage from './pages/HomePage'
import ParamPage from './pages/ParamPage'
import RapportPage from './pages/RapportPage'
import RoomPage from './pages/RoomPage'
import ServicePage from './pages/ServicePage'

function AppContent() {
  
  return (
    <div style={{ display: 'flex', marginTop: '32px', height: 'calc(100vh - 32px)' }}>
      <TiltleBar />
      <Sidebar />
      <main style={{ flex: 1, padding: '20px' }}>
        <Routes>
          {/* Page auth*/}
          <Route path="/login" element={<LoginRoute />} />
    
          {/* Page d'accueil*/}
          <Route path="/" element={<HomePage />} />
    
          {/* Page clients détai*/}
          <Route path="/clients" element={<ClientPage />} />
    
          {/* Page Comptablité*/}
          <Route path="/accounting" element={<AccountingPage />} />
    
          {/* Page Reservation*/}
          <Route path="/booking" element={<BookingPage />} />
    
          {/* Page Communication*/}
          <Route path="/communication" element={<CommunicationPage />} />
    
          {/* Page Paramètre*/}
          <Route path="/param" element={<ParamPage />} />
    
          {/* Page Rapport*/}
          <Route path="/rapports" element={<RapportPage />} />
    
          {/* Page Chambres détails*/}
          <Route path="/rooms" element={<RoomPage />} />
    
          {/* Page Services*/}
          <Route path="/services" element={<ServicePage />} />
    
          <Route path="*" element={<Navigate to="/login" replace />} />
        </Routes>
      </main>
    </div>
  )
}

function App() {

  return (
    <HashRouter>
      <AppContent />
    </HashRouter>
  )
}

export default App
